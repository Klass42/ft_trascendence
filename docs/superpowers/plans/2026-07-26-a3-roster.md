> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Wire the course roster ("Alumnos" tab) into the frontend — teachers see enrolled students in a table, change roles, expel students, and add new students by email.

**Architecture:** Two new backend endpoints (`GET /users/by-ids` + `GET /users/lookup`) in Jorge's users module provide user display data for the roster (the enrollments module is off-limits). The frontend adds a `<Tabs>` wrapper to the existing `CourseDetailPage`, conditionally showing an "Alumnos" tab for staff. The `RosterPage` component merges enrollment data with batch-fetched user data and renders a table with role `<Select>`, progress display, and double-press expel.

**Tech Stack:** NestJS 10 (backend), React 19 + Vite 6 (frontend), Prisma (data layer), design system (`@/components/ui`), CSS modules, Playwright (E2E).

  async findByIds(ids: string[], organizationId: string) {
    if (ids.length === 0) return [];
    return this.prisma.user.findMany({
      where: { id: { in: ids }, organizationId },
      select: { id: true, name: true, email: true, avatarUrl: true },
      orderBy: { name: 'asc' },
    });
  }

  async lookupByEmail(email: string, organizationId: string, callerRole: string) {
    const user = await this.prisma.user.findFirst({
      where: { organizationId, email },
      select: { id: true, name: true, email: true, avatarUrl: true, role: true },
    });
    if (!user) throw new NotFoundException('User not found');
    if (callerRole === 'TEACHER' && user.role === 'ADMIN') {
      throw new ForbiddenException('Cannot look up admin users');
    }
    return { userId: user.id, name: user.name, email: user.email, avatarUrl: user.avatarUrl };
  }
```

import { useState, useEffect, useCallback } from 'react';
import { apiClient } from '../client';
import { useAuth } from '../auth-context';
import type { EnrollmentDTO } from '../types';

interface UseEnrollmentsResult {
  data: EnrollmentDTO[] | null;
  isLoading: boolean;
  error: Error | null;
  refetch: () => void;
}

export function useEnrollments(courseId: string): UseEnrollmentsResult {
  const { user } = useAuth();
  const [data, setData] = useState<EnrollmentDTO[] | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);
  const [nonce, setNonce] = useState(0);

  const refetch = useCallback(() => setNonce((n) => n + 1), []);

  useEffect(() => {
    if (!user) return;
    let cancelled = false;
    setIsLoading(true);
    setError(null);

    apiClient
      .get<EnrollmentDTO[]>(
        `/courses/${courseId}/enrollments?organizationId=${user.organizationId}`,
      )
      .then((result) => {
        if (!cancelled) setData(result);
      })
      .catch((err) => {
        if (!cancelled) setError(err as Error);
      })
      .finally(() => {
        if (!cancelled) setIsLoading(false);
      });

    return () => { cancelled = true; };
  }, [courseId, nonce, user]);

  return { data, isLoading, error, refetch };
}
```

import { useState, useEffect } from 'react';
import { apiClient } from '../client';
import type { RosterUserDTO } from '@/features/courses/types';

interface UseUsersByIdsResult {
  data: RosterUserDTO[] | null;
  isLoading: boolean;
  error: Error | null;
}

/** Batch-fetch user display data for the roster. Empty ids = empty array, no request. */
export function useUsersByIds(ids: string[]): UseUsersByIdsResult {
  const [data, setData] = useState<RosterUserDTO[] | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<Error | null>(null);
  const key = ids.join(',');

  useEffect(() => {
    if (ids.length === 0) {
      setData([]);
      setIsLoading(false);
      return;
    }
    let cancelled = false;
    setIsLoading(true);
    setError(null);

    apiClient
      .get<RosterUserDTO[]>(`/users/by-ids?ids=${encodeURIComponent(key)}`)
      .then((result) => {
        if (!cancelled) setData(result);
      })
      .catch((err) => {
        if (!cancelled) setError(err as Error);
      })
      .finally(() => {
        if (!cancelled) setIsLoading(false);
      });

    return () => { cancelled = true; };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [key]);

  return { data, isLoading, error };
}

```tsx
import { useState, useEffect, useRef, useCallback, useMemo } from 'react';
import { apiClient, useAuth } from '@/lib/api';
import { useEnrollments, useUsersByIds } from '@/lib/api/hooks';
import { Button, Modal, Input, Select, Spinner, EmptyState, useToast } from '@/components/ui';
import type { RosterRow, RosterUserDTO } from './types';
import styles from './RosterPage.module.css';

interface RosterPageProps {
  courseId: string;
}

const ROLE_OPTIONS = [
  { value: 'STUDENT', label: 'Alumno' },
  { value: 'ASSISTANT', label: 'Asistente' },
];

export function RosterPage({ courseId }: RosterPageProps) {
  const { user } = useAuth();
  const { toast } = useToast();
  const orgId = user?.organizationId ?? '';

  // ── Data fetching ──
  const { data: enrollments, isLoading: enrollLoading, error: enrollError, refetch } = useEnrollments(courseId);

  const userIds = useMemo(
    () => (enrollments ?? []).map((e) => e.userId),
    [enrollments],
  );
  const { data: users, isLoading: usersLoading, error: usersError } = useUsersByIds(userIds);

  // ── Merge into RosterRow[] ──
  const rows: RosterRow[] = useMemo(() => {
    if (!enrollments || !users) return [];
    const userMap = new Map<string, RosterUserDTO>();
    for (const u of users) userMap.set(u.id, u);
    return enrollments
      .filter((e) => e.role !== 'TEACHER') // Don't show course teachers in the roster
      .map((e) => {
        const u = userMap.get(e.userId);
        return {
          enrollmentId: e.id,
          userId: e.userId,
          name: u?.name ?? '—',
          email: u?.email ?? '',
          avatarUrl: u?.avatarUrl ?? null,
          role: e.role,
          progress: e.progress,
        };
      });
  }, [enrollments, users]);

  // ── Role change ──
  const [changingRole, setChangingRole] = useState<string | null>(null);

  const handleRoleChange = useCallback(async (enrollmentId: string, newRole: string) => {
    setChangingRole(enrollmentId);
    try {
      await apiClient.patch(`/courses/${courseId}/enrollments/${enrollmentId}/role?organizationId=${orgId}`, { role: newRole });
      toast({ message: 'Rol actualizado', variant: 'success' });
      refetch();
    } catch {
      toast({ message: 'Error al cambiar el rol', variant: 'error' });
    } finally {
      setChangingRole(null);
    }
  }, [courseId, orgId, toast, refetch]);

  // ── Double-press expel ──
  const [expellingId, setExpellingId] = useState<string | null>(null);
  const expelTimer = useRef<ReturnType<typeof setTimeout> | null>(null);

  useEffect(() => {
    return () => {
      if (expelTimer.current) clearTimeout(expelTimer.current);
    };
  }, []);

  const startExpel = useCallback((userId: string) => {
    setExpellingId(userId);
    expelTimer.current = setTimeout(() => {
      setExpellingId(null);
      expelTimer.current = null;
    }, 5000);
  }, []);

  const confirmExpel = useCallback(async (userId: string) => {
    if (expelTimer.current) {
      clearTimeout(expelTimer.current);
      expelTimer.current = null;
    }
    setExpellingId(null);
    try {
      await apiClient.delete(`/courses/${courseId}/enroll?organizationId=${orgId}&userId=${userId}`);
      toast({ message: 'Alumno expulsado', variant: 'success' });
      refetch();
    } catch {
      toast({ message: 'Error al expulsar alumno', variant: 'error' });
    }
  }, [courseId, orgId, toast, refetch]);

  const cancelExpel = useCallback(() => {
    if (expelTimer.current) {
      clearTimeout(expelTimer.current);
      expelTimer.current = null;
    }
    setExpellingId(null);
  }, []);

  // ── Add student modal ──
  const [addOpen, setAddOpen] = useState(false);
  const [searchEmail, setSearchEmail] = useState('');
  const [lookupResult, setLookupResult] = useState<{ userId: string; name: string; email: string; avatarUrl: string | null } | null>(null);
  const [lookupLoading, setLookupLoading] = useState(false);
  const [lookupError, setLookupError] = useState<string>();
  const [enrolling, setEnrolling] = useState(false);

  const handleSearch = useCallback(async () => {
    if (!searchEmail.trim()) return;
    setLookupLoading(true);
    setLookupError(undefined);
    setLookupResult(null);
    try {
      const result = await apiClient.get<{ userId: string; name: string; email: string; avatarUrl: string | null }>(
        `/users/lookup?email=${encodeURIComponent(searchEmail.trim())}`,
      );
      setLookupResult(result);
    } catch (err) {
      const msg = (err as { status?: number })?.status === 404
        ? 'No se encontró un usuario con ese email'
        : (err as { status?: number })?.status === 403
          ? 'No tienes permisos para ver este usuario'
          : 'Error al buscar usuario';
      setLookupError(msg);
    } finally {
      setLookupLoading(false);
    }
  }, [searchEmail]);

  const handleEnroll = useCallback(async () => {
    if (!lookupResult || !user) return;
    setEnrolling(true);
    try {
      await apiClient.post(
        `/courses/${courseId}/enroll?organizationId=${orgId}&userId=${lookupResult.userId}`,
      );
      toast({ message: `${lookupResult.name} inscrito al curso`, variant: 'success' });
      setAddOpen(false);
      setSearchEmail('');
      setLookupResult(null);
      refetch();
    } catch (err) {
      const msg = (err as { status?: number })?.status === 409
        ? 'Este usuario ya está inscrito en el curso'
        : 'Error al inscribir alumno';
      toast({ message: msg, variant: 'error' });
    } finally {
      setEnrolling(false);
    }
  }, [lookupResult, user, courseId, orgId, toast, refetch]);

  const closeModal = useCallback(() => {
    setAddOpen(false);
    setSearchEmail('');
    setLookupResult(null);
    setLookupError(undefined);
  }, []);

  // ── Loading / Error / Empty states ──
  const isLoading = enrollLoading || usersLoading;
  const error = enrollError || usersError;

  if (isLoading) {
    return (
      <div className={styles.loadingWrap}>
        <Spinner size="lg" label="Cargando alumnos..." />
      </div>
    );
  }

  if (error) {
    return (
      <div className={styles.page}>
        <EmptyState
          title="Error al cargar alumnos"
          description="No se pudo obtener la lista de inscripciones."
          action={<Button variant="secondary" size="sm" onClick={refetch}>Reintentar</Button>}
        />
      </div>
    );
  }

  return (
    <div className={styles.page}>
      <div className={styles.header}>
        <h2 className={styles.title}>Alumnos inscritos</h2>
        <Button variant="primary" size="sm" onClick={() => setAddOpen(true)}>
          + Añadir alumno
        </Button>
      </div>

      {rows.length === 0 ? (
        <EmptyState
          title="No hay alumnos inscritos"
          description="Añade estudiantes al curso usando el botón superior."
        />
      ) : (
        <table className={styles.table}>
          <thead>
            <tr>
              <th className={styles.th}>Alumno</th>
              <th className={`${styles.th} ${styles.emailHeader}`}>Email</th>
              <th className={styles.th}>Rol</th>
              <th className={`${styles.th} ${styles.progressHeader}`}>Progreso</th>
              <th className={styles.th}>Acción</th>
            </tr>
          </thead>
          <tbody>
            {rows.map((row) => (
              <tr key={row.enrollmentId} className={styles.tr}>
                <td className={styles.td}>
                  <div className={styles.nameCell}>
                    {row.avatarUrl ? (
                      <img src={row.avatarUrl} alt="" className={styles.avatar} />
                    ) : (
                      <span className={styles.avatarPlaceholder}>
                        {row.name.charAt(0).toUpperCase()}
                      </span>
                    )}
                    {row.name}
                  </div>
                </td>
                <td className={`${styles.td} ${styles.emailCell}`}>{row.email}</td>
                <td className={styles.td}>
                  <div className={styles.roleSelect}>
                    <Select
                      value={row.role}
                      options={ROLE_OPTIONS}
                      disabled={changingRole === row.enrollmentId}
                      onChange={(e) => handleRoleChange(row.enrollmentId, e.target.value)}
                      fullWidth={false}
                    />
                  </div>
                </td>
                <td className={`${styles.td} ${styles.progressCell}`}>
                  {row.progress} %
                </td>
                <td className={styles.td}>
                  {expellingId === row.userId ? (
                    <div className={styles.dangerActions}>
                      <Button variant="danger" size="sm" onClick={() => confirmExpel(row.userId)}>
                        Confirmar expulsión
                      </Button>
                      <Button variant="ghost" size="sm" onClick={cancelExpel}>
                        Cancelar
                      </Button>
                    </div>
                  ) : (
                    <Button variant="ghost" size="sm" onClick={() => startExpel(row.userId)}>
                      Expulsar
                    </Button>
                  )}
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      )}

      {/* Add student modal */}
      <Modal open={addOpen} onClose={closeModal} title="Añadir alumno">
        <div style={{ display: 'flex', gap: 8, alignItems: 'flex-end' }}>
          <div style={{ flex: 1 }}>
            <Input
              label="Email del alumno"
              value={searchEmail}
              onChange={(e) => setSearchEmail(e.target.value)}
              placeholder="alumno@ejemplo.com"
              onKeyDown={(e) => { if (e.key === 'Enter') handleSearch(); }}
            />
          </div>
          <Button
            variant="secondary"
            size="sm"
            onClick={handleSearch}
            loading={lookupLoading}
            style={{ marginBottom: 2 }}
          >
            Buscar
          </Button>
        </div>

        {lookupError && <p className={styles.modalError}>{lookupError}</p>}

        {lookupResult && (
          <div className={styles.modalResult}>
            {lookupResult.avatarUrl ? (
              <img src={lookupResult.avatarUrl} alt="" className={styles.avatar} />
            ) : (
              <span className={styles.avatarPlaceholder}>
                {lookupResult.name.charAt(0).toUpperCase()}
              </span>
            )}
            <div style={{ flex: 1 }}>
              <div style={{ fontWeight: 600, fontSize: 14 }}>{lookupResult.name}</div>
              <div style={{ fontSize: 13, color: 'var(--color-text-secondary)' }}>{lookupResult.email}</div>
            </div>
            <Button variant="primary" size="sm" onClick={handleEnroll} loading={enrolling}>
              Añadir al curso
            </Button>
          </div>
        )}
      </Modal>
    </div>
  );
}
```

import { test, expect } from '@playwright/test';
import { actAs, TEACHER, STUDENT } from './dev-user';

test.describe('Roster (Alumnos tab)', () => {

    await actAs(page, TEACHER);
  });

    await page.getByText('Introducción al Desarrollo Web').first().click({ timeout: 8000 });
    await expect(page).toHaveURL(/\/courses\//);

    // Tabs should be visible for teacher
    const alumnosTab = page.getByRole('tab', { name: 'Alumnos' });
    await expect(alumnosTab).toBeVisible({ timeout: 8000 });

    await page.screenshot({ path: 'e2e/screenshots/roster-tab-visible.png' });
  });

  test('student does NOT see "Alumnos" tab', async ({ page }) => {
    await actAs(page, STUDENT);
    await page.goto('/courses');
    await page.getByText('Introducción al Desarrollo Web').first().click({ timeout: 8000 });
    await expect(page).toHaveURL(/\/courses\//);

    // Student should not see the tab at all
    await expect(page.getByRole('tab', { name: 'Alumnos' })).toHaveCount(0);
  });

  test('teacher views roster table with enrolled students', async ({ page }) => {
    await page.goto('/courses');
    await page.getByText('Introducción al Desarrollo Web').first().click({ timeout: 8000 });

    const alumnosTab = page.getByRole('tab', { name: 'Alumnos' });
    await alumnosTab.click();

    // Wait for the table to appear
    const table = page.locator('table');
    await expect(table).toBeVisible({ timeout: 8000 });

    // Should show at least the seeded student
    await expect(table.getByText('Lucía')).toBeVisible();

    await page.screenshot({ path: 'e2e/screenshots/roster-table.png' });
  });

  test('teacher changes student role to ASSISTANT', async ({ page }) => {
    await page.goto('/courses');
    await page.getByText('Introducción al Desarrollo Web').first().click({ timeout: 8000 });
    await page.getByRole('tab', { name: 'Alumnos' }).click();

    const table = page.locator('table');
    await expect(table).toBeVisible({ timeout: 8000 });

    // Find the row with the seeded student and change role
    const row = table.locator('tr', { has: page.getByText('Lucía') });
    const select = row.locator('select');
    await select.selectOption('ASSISTANT');

    // Toast should appear confirming the change
    await expect(page.getByText('Rol actualizado')).toBeVisible({ timeout: 5000 });

    await page.screenshot({ path: 'e2e/screenshots/roster-role-change.png' });
  });

  test('teacher expels student (double press)', async ({ page }) => {
    await page.goto('/courses');
    await page.getByText('Introducción al Desarrollo Web').first().click({ timeout: 8000 });
    await page.getByRole('tab', { name: 'Alumnos' }).click();

    const table = page.locator('table');
    await expect(table).toBeVisible({ timeout: 8000 });

    // First press → shows "Confirmar expulsión"
    const row = table.locator('tr', { has: page.getByText('Lucía') });
    await row.getByRole('button', { name: 'Expulsar' }).click();
    await expect(row.getByRole('button', { name: 'Confirmar expulsión' })).toBeVisible();

    // Second press → expels
    await row.getByRole('button', { name: 'Confirmar expulsión' }).click();
    await expect(page.getByText('Alumno expulsado')).toBeVisible({ timeout: 5000 });

    // Student should no longer appear
    await expect(table.getByText('Lucía')).toHaveCount(0, { timeout: 5000 });

    await page.screenshot({ path: 'e2e/screenshots/roster-expel.png' });
  });
});
